{{- $image := .image -}} {{- if and (not $image) .src -}} {{- $image = resources.Get .src -}} {{- end -}} {{- if not $image -}} {{- errorf "Image resource %q not found" .src -}} {{- else -}} {{- $widths := .widths | default (slice 480 960) -}} {{- $webpSrcset := slice -}} {{- range $width := $widths -}} {{- if le $width $image.Width -}} {{- $webp := $image.Resize (printf "%dx webp q80" $width) -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" $webp.RelPermalink $webp.Width) -}} {{- end -}} {{- end -}} {{- if not $webpSrcset -}} {{- $webp := $image.Resize (printf "%dx webp q80" $image.Width) -}} {{- $webpSrcset = $webpSrcset | append (printf "%s %dw" $webp.RelPermalink $webp.Width) -}} {{- end -}} {{- $fallbackWidth := .fallbackWidth | default 960 -}} {{- if gt $fallbackWidth $image.Width -}} {{- $fallbackWidth = $image.Width -}} {{- end -}} {{- $fallback := $image.Resize (printf "%dx jpg q84" $fallbackWidth) -}} {{- end -}}
